// SPDX-License-Identifier: MIT pragma solidity ^0.8.19; contract PokeBetArena { struct Match { bytes32 matchId; address player1; address player2; uint256 stake1; uint256 stake2; address winner; string psRoomId; bool settled; } address public oracle; mapping(bytes32 => Match) public matches; uint256 public totalVolume; event MatchCreated(bytes32 id, address p1, address p2); event BetPlaced(bytes32 id, address better, uint256 amount); event MatchSettled(bytes32 id, address winner, uint256 payout); modifier onlyOracle() { require(msg.sender == oracle, "Not oracle"); _; } function createMatch(address p2, string calldata roomId) external payable returns (bytes32) { bytes32 id = keccak256(abi.encodePacked( msg.sender, p2, roomId, block.timestamp)); matches[id] = Match(id, msg.sender, p2, msg.value, 0, address(0), roomId, false); totalVolume += msg.value; emit MatchCreated(id, msg.sender, p2); return id; } function joinMatch(bytes32 matchId) external payable { Match storage m = matches[matchId]; require(msg.value == m.stake1, "Stake mismatch"); m.stake2 = msg.value; totalVolume += msg.value; emit BetPlaced(matchId, msg.sender, msg.value); } function settleMatch( bytes32 matchId, address winner ) external onlyOracle { Match storage m = matches[matchId]; require(!m.settled, "Already settled"); m.winner = winner; m.settled = true; uint256 payout = m.stake1 + m.stake2; payable(winner).transfer(payout); emit MatchSettled(matchId, winner, payout); } }
